    {\large\bfseries\color{cherry}Generic codecs miss the structure}\\[0.18cm]
    {\normalsize\color{slate}\raggedright
     gzip and Zstd match over local byte windows, so cross-haplotype repeats go
     unexploited. Specialized tools drop optional fields or run at a few MB/s.\par}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.315\linewidth}
    {\large\bfseries\color{cherry}Analysis pays the cost twice}\\[0.18cm]
    {\normalsize\color{slate}\raggedright
     \code{vg}, \code{odgi} and Panacus must decompress \emph{and} rebuild the whole
     graph before answering one query. That costs hundreds of GB of RAM.\par}
  \end{minipage}
\end{minipage}
\end{center}

\begin{panel}{A lossless round trip through one container}
  \centering
  \begin{tikzpicture}[

      ar/.style={-{Stealth[length=5mm,width=4mm]},line width=3pt,color=slate},
      tag/.style={font=\small\bfseries,inner sep=2pt}]

    \node[end] (gfa) at (3.4,0)
      {{\bfseries\large GFA text}\\[0.18cm]
       {\small S \textperiodcentered\ L \textperiodcentered\ P\\
        W \textperiodcentered\ J \textperiodcentered\ C\\[2pt]optional fields}};
    \node[end,draw=cherry,fill=palec] (gz) at (42.6,0)
      {{\bfseries\large\color{cherry}\code{.gfaz}}\\[0.18cm]
       {\small one container\\for CPU + GPU}};

    \node[st]  (t1) at (12.8,2.25) {\textbf{Parallel parse}\\[1pt]{\small typed column store}};
    \node[hot] (t2) at (23.0,2.25) {\textbf{2-mer grammar}\\[1pt]{\small on paths + walks}};
    \node[st]  (t3) at (33.2,2.25) {\textbf{Field codecs}\\[1pt]{\small varint + Zstd}};

    \node[st]  (b1) at (12.8,-2.25) {\textbf{Rebuild + write}\\[1pt]{\small byte-identical GFA}};
    \node[hot] (b2) at (23.0,-2.25) {\textbf{Expand rules}\\[1pt]{\small + inverse delta}};
    \node[st]  (b3) at (33.2,-2.25) {\textbf{Column decode}\\[1pt]{\small streamed, parallel}};

    \draw[ar] (gfa.east|-t1) -- (t1.west);
    \draw[ar] (t1) -- (t2);  \draw[ar] (t2) -- (t3);
    \draw[ar] (t3.east) -- (gz.west|-t3);
    \draw[ar] (gz.west|-b3) -- (b3.east);
    \draw[ar] (b3) -- (b2);  \draw[ar] (b2) -- (b1);
    \draw[ar] (b1.west) -- (gfa.east|-b1);

    \node[tag,color=cherry,anchor=south] at (23.0,4.0) {COMPRESS $\rightarrow$};
    \node[tag,color=teal,anchor=north]   at (23.0,-4.0) {$\leftarrow$ DECOMPRESS};
  \end{tikzpicture}

  \cpt{Both directions are \hd{fully parallel} (OpenMP; a CUDA backend shares the
  same format). Columns are varint- and bit-packed before Zstd. Optional fields and
  record order survive intact, so \code{.gfaz} \emph{replaces} the GFA rather than
  shadowing it.}
\end{panel}

\begin{panel}{The idea: a recursive grammar of 2-mers}
  {\normalsize\color{slate}\raggedright
   Haplotypes repeat each other. \gz{} replaces the most frequent \emph{adjacent
   pair} of symbols with a rule, then repeats. Later rounds reuse earlier rules, so
   each round captures longer repeats.\par}
  \vspace{0.26cm}
  \centering
  \colorbox{paleg}{\parbox{0.96\linewidth}{
    \centering\vspace{0.26cm}
    \renewcommand{\arraystretch}{1.18}
    \begin{tabular}{>{\bfseries}p{5.2cm} p{10.4cm} l}
      \textcolor{slate}{Input} & \textcolor{muted}{no rules yet} & \code{12 7 12 7 12 7 12 7 31} \\
      \textcolor{cherry}{Round 1} & $R_1 \leftarrow (12,7)$ & \code{$R_1$ $R_1$ $R_1$ $R_1$ 31} \\
      \textcolor{cherry}{Round 2} & $R_2 \leftarrow (R_1,R_1)$ & \code{$R_2$ $R_2$ 31} \\
      \textcolor{teal}{Round 3} & $R_3 \leftarrow (R_2,R_2)$ & \code{$R_3$ 31} \\
    \end{tabular}
    \vspace{0.26cm}}}
  \cpt{Each round is one count-and-rewrite scan: \hd{$O(N)$ per round, fixed round
  count}, so compression is linear and parallel across chunks. Paths and walks share
  one rulebook. Orientations are canonicalized, so a forward repeat also matches its
  reverse complement.}
\end{panel}

\begin{panel}{CPU performance against five baselines}
  \centering
  \includegraphics[width=\linewidth]{cpu_ratio.pdf}
  \cpt{\hd{\gz{} compresses better than every baseline on all six graphs}, reaching
  84$\times$ on HPRC v2.0: 20$\times$ past gzip, 13$\times$ past Zstd.}
  \vspace{0.22cm}
  \includegraphics[width=\linewidth]{cpu_throughput.pdf}
  \cpt{End-to-end CLI wall time, 32 threads; same legend as above. \hd{\gz{}

\end{panel}

    \node[stage,draw=cherry,line width=3pt,fill=palec,right=of compute] (result)
      {{\small\bfseries\color{cherry}RESULT}\\[0.34cm]
       {\bfseries VCF \textperiodcentered\ TSV}\\[0.24cm]

    \draw[ar] (packed) -- (extract);

    \node[fill=slate,text=white,rounded corners=8pt,minimum height=1.8cm,
          text width=42.6cm,align=center,font=\normalsize\bfseries,
          below=7mm of $(extract.south)!0.5!(compute.south)$] (memory)
      {one selected stream \ $+$ \ bounded decoder scratch \ $+$ \ a compact
       accumulator};
    \node[anchor=north,font=\small\bfseries,text=cherry,below=3mm of memory]
      {UNUSED STREAMS STAY COMPRESSED \ \ $\cdot$ \ \ THE FULL GFA IS NEVER MATERIALIZED};
  \end{tikzpicture}

  \cpt{Most analyses only \emph{read} traversals, folding node visits into a small
  accumulator. \gz{} runs that fold on the grammar-compressed columns, so \hd{peak
  memory stays below the original GFA}.}
\end{panel}

\begin{panel}{Six analyses, one decoder}
  \centering
  {\normalsize
  \setlength{\tabcolsep}{10pt}\renewcommand{\arraystretch}{1.14}
  \begin{tabular}{llcl}
